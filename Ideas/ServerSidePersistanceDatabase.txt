Cloud-Only Alternative (No MySQL on gameserver)
// POST to your secure HTTPS endpoint (curl)
void SavePlayerREST(uint64 steamid, const json& data) {
    std::string payload = data.dump();
    curl_easy_setopt(curl, CURLOPT_URL, "https://api.crimewars.net/v1/save");
    curl_easy_setopt(curl, CURLOPT_POSTFIELDS, payload.c_str());
    curl_easy_perform(curl);  // Async or fire-and-forget
}
