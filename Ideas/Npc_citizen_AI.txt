// npc_citizen.cpp: Curl callback (static)
static size_t WriteCallback(void* contents, size_t size, size_t nmemb, void* userp) {
    ((std::string*)userp)->append((char*)contents, size * nmemb);
    return size * nmemb;
}

// QueryAI: POST to Grok API (alt: Gemini "https://generativelanguage.googleapis.com/v1beta/models/gemini-pro:generateContent?key=API_KEY")
void CNPC_Citizen::QueryAI(const char* prompt, std::string& response) {
    CURL* curl = curl_easy_init();
    if (curl) {
        std::string url = "https://api.x.ai/v1/chat/completions";  // Grok
        std::string body = "{\"model\": \"grok-beta\", \"messages\": [{\"role\": \"user\", \"content\": \"" + std::string(prompt) + "\"}]}";
        struct curl_slist* headers = NULL;
        headers = curl_slist_append(headers, "Content-Type: application/json");
        headers = curl_slist_append(headers, ("Authorization: Bearer " + std::string(m_strAPIKey)).c_str());

        curl_easy_setopt(curl, CURLOPT_URL, url.c_str());
        curl_easy_setopt(curl, CURLOPT_POSTFIELDS, body.c_str());
        curl_easy_setopt(curl, CURLOPT_HTTPHEADER, headers);
        curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, WriteCallback);
        curl_easy_setopt(curl, CURLOPT_WRITEDATA, &response);

        CURLcode res = curl_easy_perform(curl);
        if (res != CURLE_OK) { Warning("Curl failed: %s\n", curl_easy_strerror(res)); }

        curl_slist_free_all(headers);
        curl_easy_cleanup(curl);
    }
}
Dynamic Interactions: In SpeakTL/HandleAnimEvent, query AI for responses.
void CNPC_Citizen::DynamicSpeak(const char* playerInput) {
    std::string prompt = "As a " + GetFaction() + " NPC in urban chaos MMO: respond to player: " + std::string(playerInput);
    std::string resp;
    QueryAI(prompt.c_str(), resp);
    // Parse JSON resp (use nlohmann/json or manual): e.g., choices[0].message.content
    size_t pos = resp.find("\"content\":\"");
    if (pos != std::string::npos) {
        std::string aiText = resp.substr(pos + 11, resp.find("\"", pos + 11) - pos - 11);
        Speak(aiText.c_str());  // Or EmitSound/Subtitle
    }
}

int CNPC_Citizen::SelectSchedule() {
    if (m_pSquad && m_pSquad->NumMembers() > 1) {
        std::string prompt = "Squad decision: " + GetSquadState() + " in MMO world. Action? (e.g., patrol/attack/flank)";
        std::string resp;
        QueryAI(prompt.c_str(), resp);
        // Parse to SCHED_ (e.g., if "attack" return SCHED_COMBAT_FACE)
        if (resp.find("attack") != std::string::npos) return SCHED_COMBAT_FACE;
    }
    return BaseClass::SelectSchedule();
}
